1. SELECT marka, model, cena FROM pojazdy WHERE marka LIKE '%BM%' ORDER BY cena DESC LIMIT 15;

2.SELECT AVG(cena) as 'Średnia cena', MAX(cena) as 'Maksymalna cena' FROM pojazdy WHERE model = "meta";

3.SELECT pojazdy.marka, pojazdy.model, pojazdy.cena, kolory.nazwa, kolory.doplata FROM pojazdy JOIN kolory ON pojazdy.kolor = kolory.id WHERE model = "alfa";

4. SELECT marka, model, cena FROM pojazdy ORDER BY RAND() LIMIT 2;
